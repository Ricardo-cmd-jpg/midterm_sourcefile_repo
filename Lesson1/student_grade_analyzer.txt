let scores = [85, 72, 91, 64, 78];

console.log("Student Scores:");

for (let score of scores) {
    console.log(score);
}

console.log("");

// Variables
let total = 0;
let highest = scores[0];
let lowest = scores[0];
let passed = 0;
let failed = 0;

// Analyze each student's score
for (let i = 0; i < scores.length; i++) {
    let score = scores[i];
    let rating = "";
    let status = "";

    // Determine rating and status
    if (score >= 90 && score <= 100) {
        rating = "Excellent";
        status = "Passed";
    } 
    else if (score >= 80 && score <= 89) {
        rating = "Very Good";
        status = "Passed";
    } 
    else if (score >= 75 && score <= 79) {
        rating = "Good";
        status = "Passed";
    } 
    else {
        status = "Failed";
    }

    // Display student result
    if (rating !== "") {
        console.log(
            "Student " + (i + 1) + ": " + score + " - " + rating + " - " + status
        );
    } 
    else {
        console.log(
            "Student " + (i + 1) + ": " + score + " - " + status
        );
    }

    // Calculate total
    total += score;

    // Find highest and lowest
    if (score > highest) {
        highest = score;
    }

    if (score < lowest) {
        lowest = score;
    }

    // Count passed and failed
    if (score >= 75) {
        passed++;
    } 
    else {
        failed++;
    }
}

// Calculate average
let average = total / scores.length;

// Display results
console.log("");
console.log("Total Score: " + total);
console.log("Average Score: " + average);
console.log("Highest Score: " + highest);
console.log("Lowest Score: " + lowest);
console.log("Passed: " + passed);
console.log("Failed: " + failed);
